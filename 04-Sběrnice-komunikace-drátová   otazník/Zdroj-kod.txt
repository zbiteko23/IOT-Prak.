from machine import Pin, UART
from time import sleep

uart = UART(0, 9600, tx=Pin(0))
tl = Pin(15, Pin.IN, Pin.PULL_UP)

while True:
    uart.write("1" if tl.value() == 0 else "0")
    sleep(0.1)

from machine import Pin, UART

uart = UART(0, 9600, rx=Pin(1))
led = Pin(16, Pin.OUT)

while True:
    if uart.any():
        data = uart.read()
        if data:
            led.value(data[-1:] == b"1")
