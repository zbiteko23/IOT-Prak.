import smbus, time
 
ADDR = 0x27          # pokud i2cdetect ukazuje 3f, dejte 0x3F
bus = smbus.SMBus(1)
 
def w(b):
    bus.write_byte(ADDR, b | 0x08)      # 0x08 = podsvit
 
def nib(n, rs):
    d = (n << 4) | rs
    w(d | 4)
    time.sleep(0.001)
    w(d & ~4)
    time.sleep(0.001)
 
def send(v, rs=0):
    nib(v >> 4, rs)
    nib(v & 15, rs)
 
def init():
    time.sleep(0.05)
    for n in (3, 3, 3, 2):
        nib(n, 0)
        time.sleep(0.005)
    send(0x28)
    send(0x0C)
    send(0x06)
    send(0x01)
    time.sleep(0.005)
 
def text(s, row=0):
    send(0x80 + (0x40 if row else 0))
    for c in s.ljust(16)[:16]:
        send(ord(c), 1)
 
init()
text("Ahoj svete!", 0)
text("Raspberry Pi 4", 1)
